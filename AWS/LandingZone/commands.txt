# $1 account id, $2 service principal
register_admin() {  
  aws organizations register-delegated-administrator \
    --account-id "$1" --service-principal "$2"
}

# DenyRootUser: denies every action performed by the root user of member accounts
aws organizations create-policy \
  --name DenyRootUser \
  --description "Denies actions by the root user" \
  --type SERVICE_CONTROL_POLICY \
  --content '{
    "Version": "2012-10-17",
    "Statement": [
      {
        "Sid": "DenyRootUser",
        "Effect": "Deny",
        "Action": "*",
        "Resource": "*",
        "Condition": {
          "StringLike": {
            "aws:PrincipalArn": "arn:aws:iam::*:root"
          }
        }
      }
    ]
  }'
 
# RestrictRegions: denies everything outside the allowed regions (global services excluded via NotAction)
aws organizations create-policy \
  --name RestrictRegions \
  --description "Restricts usage to the allowed regions" \
  --type SERVICE_CONTROL_POLICY \
  --content '{
    "Version": "2012-10-17",
    "Statement": [
      {
        "Sid": "DenyOutsideAllowedRegions",
        "Effect": "Deny",
        "NotAction": [
          "iam:*",
          "organizations:*",
          "sts:*",
          "sso:*",
          "identitystore:*",
          "account:*",
          "support:*",
          "budgets:*",
          "ce:*",
          "health:*",
          "trustedadvisor:*",
          "route53:*",
          "route53domains:*",
          "cloudfront:*",
          "waf:*",
          "wafv2:*"
        ],
        "Resource": "*",
        "Condition": {
          "StringNotEquals": {
            "aws:RequestedRegion": [
              "us-east-1"
            ]
          }
        }
      }
    ]
  }'

# Setup AWS Config baseline as follows:
#
# Log Archive -> stores the Config snapshots/history (bucket)
# Shared      -> deploys the Config recorder + delivery channel to every
#                account through a service-managed StackSet
#                (it is the delegated administrator for StackSets)
# Management  -> not covered by service-managed StackSets, so it gets its
#                own stack

# In Management account
export ORG_ID=$(aws organizations describe-organization --query 'Organization.Id' --output text)

# Bucket name (the account ID makes it unique)
export CONFIG_BUCKET=org-config-logs-$ACCOUNT_LOG_ARCHIVE

# Template deployed to every account: IAM role + recorder + delivery channel
# (quoted 'EOF' so the shell does not touch anything inside the template)
cat > /tmp/config-baseline.json <<'EOF'
{
  "AWSTemplateFormatVersion": "2010-09-09",
  "Description": "AWS Config baseline: recorder and delivery channel",
  "Parameters": {
    "ConfigBucketName": {
      "Type": "String"
    }
  },
  "Resources": {
    "ConfigRole": {
      "Type": "AWS::IAM::Role",
      "Properties": {
        "AssumeRolePolicyDocument": {
          "Version": "2012-10-17",
          "Statement": [
            {
              "Effect": "Allow",
              "Principal": {"Service": "config.amazonaws.com"},
              "Action": "sts:AssumeRole"
            }
          ]
        },
        "ManagedPolicyArns": [
          "arn:aws:iam::aws:policy/service-role/AWS_ConfigRole"
        ]
      }
    },
    "DeliveryChannel": {
      "Type": "AWS::Config::DeliveryChannel",
      "Properties": {
        "Name": "default",
        "S3BucketName": {"Ref": "ConfigBucketName"}
      }
    },
    "ConfigRecorder": {
      "Type": "AWS::Config::ConfigurationRecorder",
      "DependsOn": "DeliveryChannel",
      "Properties": {
        "Name": "default",
        "RoleARN": {"Fn::GetAtt": ["ConfigRole", "Arn"]},
        "RecordingGroup": {
          "AllSupported": true,
          "IncludeGlobalResourceTypes": true
        }
      }
    }
  }
}
EOF

# Assume role in LOG_ARCHIVE account
CREDS=$(aws sts assume-role --role-arn arn:aws:iam::$ACCOUNT_LOG_ARCHIVE:role/OrganizationAccountAccessRole --role-session-name log-archive --query 'Credentials.[AccessKeyId,SecretAccessKey,SessionToken]' --output text)
export AWS_ACCESS_KEY_ID=$(echo $CREDS | awk '{print $1}')
export AWS_SECRET_ACCESS_KEY=$(echo $CREDS | awk '{print $2}')
export AWS_SESSION_TOKEN=$(echo $CREDS | awk '{print $3}')
 
# Verify
aws sts get-caller-identity

# Setup S3 bucket in LOG_ARCHIVE account for AWS Config
aws s3api create-bucket --bucket $CONFIG_BUCKET
 
# Versioning
aws s3api put-bucket-versioning --bucket $CONFIG_BUCKET --versioning-configuration Status=Enabled
 
# Block all public access
aws s3api put-public-access-block --bucket $CONFIG_BUCKET --public-access-block-configuration BlockPublicAcls=true,IgnorePublicAcls=true,BlockPublicPolicy=true,RestrictPublicBuckets=true
 
# Default encryption (SSE-S3; can be hardened later with a KMS key)
aws s3api put-bucket-encryption --bucket $CONFIG_BUCKET --server-side-encryption-configuration '{"Rules":[{"ApplyServerSideEncryptionByDefault":{"SSEAlgorithm":"AES256"}}]}'
 
# Bucket policy: only AWS Config from accounts of this organization can write
cat > /tmp/config-bucket-policy.json <<EOF
{
  "Version": "2012-10-17",
  "Statement": [
    {
      "Sid": "AWSConfigBucketPermissionsCheck",
      "Effect": "Allow",
      "Principal": {"Service": "config.amazonaws.com"},
      "Action": ["s3:GetBucketAcl", "s3:ListBucket"],
      "Resource": "arn:aws:s3:::$CONFIG_BUCKET",
      "Condition": {
        "StringEquals": {
          "aws:SourceOrgID": "$ORG_ID"
        }
      }
    },
    {
      "Sid": "AWSConfigBucketDelivery",
      "Effect": "Allow",
      "Principal": {"Service": "config.amazonaws.com"},
      "Action": "s3:PutObject",
      "Resource": "arn:aws:s3:::$CONFIG_BUCKET/AWSLogs/*/Config/*",
      "Condition": {
        "StringEquals": {
          "s3:x-amz-acl": "bucket-owner-full-control",
          "aws:SourceOrgID": "$ORG_ID"
        }
      }
    },
    {
      "Sid": "DenyNonTLS",
      "Effect": "Deny",
      "Principal": "*",
      "Action": "s3:*",
      "Resource": [
        "arn:aws:s3:::$CONFIG_BUCKET",
        "arn:aws:s3:::$CONFIG_BUCKET/*"
      ],
      "Condition": {
        "Bool": {"aws:SecureTransport": "false"}
      }
    }
  ]
}
EOF
 
# Apply the policy
aws s3api put-bucket-policy --bucket $CONFIG_BUCKET --policy file:///tmp/config-bucket-policy.json
 
# Verify the bucket
aws s3api get-bucket-policy --bucket $CONFIG_BUCKET --query Policy --output text
aws s3api get-bucket-versioning --bucket $CONFIG_BUCKET
aws s3api get-public-access-block --bucket $CONFIG_BUCKET
